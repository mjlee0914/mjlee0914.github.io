##### `practice/js/pr5-1.js` — 함수, 콜백, 클로저

- `greetUser(name, age)` 함수는 이름과 나이를 받아 인사말 문자열을 반환합니다.
- `performCalculation()`은 전달받은 콜백 함수를 두 숫자와 함께 호출합니다. `add` 또는 `multiply`를 전달하면 같은 함수로 덧셈이나 곱셈을 실행할 수 있습니다.
- `createPiggyBank()`는 `balance`를 함수 안에 보관하고, 입금 함수와 잔액 조회 함수를 객체로 반환합니다. 반환된 함수들이 바깥 함수가 끝난 뒤에도 `balance`에 접근하는 클로저 예제입니다.


```javascript
// 5-1-1

function greetUser(name, age) {
    return `안녕하세요 ${name}님! ${age}세가 되신 것을 진심으로 축하합니다.`
};

const me = greetUser('이명지', 27);
// console.log(me);


// 5-1-2
function performCalculation(num1, num2, callback) {
    return callback(num1, num2);
};


const add  = (a, b) => a + b;
const multiply = (a, b) => a * b;

//console.log(performCalculation(12, 4, add));
//console.log(performCalculation(12, 4, multiply));

// 5-1-3

function createPiggyBank(amount) {
    let balance = 0;

    // closure 적용
    // return값으로 두 개의 함수를 담은 객체 하나를 반환
    return {
        deposit(amount) {
            return balance += amount;
        },
        getBalance() {
            return balance;
        }
    }
}

const myPiggyBank = createPiggyBank();

myPiggyBank.deposit(1500);
console.log(myPiggyBank.getBalance());
```
